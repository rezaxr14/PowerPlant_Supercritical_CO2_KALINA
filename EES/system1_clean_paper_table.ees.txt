"--------------------------------------------------------------------------"
" 1) OXY-GASIFIER / GT / MIXER"
"--------------------------------------------------------------------------"

m_dot_O2 = OBWF*m_dot_bio

CGE = (0.746 - 0.18*(OBWF-0.15)^2)*(1 + 0.02*(SBWF-0.15))

mf_H2_syngas  = 0.045
mf_CO_syngas  = 0.411
mf_CH4_syngas = 0.034
mf_CO2_syngas = 0.510

m_dot_syngas = (CGE*m_dot_bio*LHV_bio)/LHV_syngas

T_syngas = 1000 [C]
h_syngas = mf_H2_syngas*Enthalpy(H2,T=T_syngas) + mf_CO_syngas*Enthalpy(CO,T=T_syngas) + mf_CH4_syngas*Enthalpy(Methane,T=T_syngas) + mf_CO2_syngas*Enthalpy(CO2,T=T_syngas)

"Gas recirculation / mixer"
m_dot_comb_out = m_dot_comp_in + m_dot_syngas
m_dot_egr      = RR*m_dot_comb_out
m_dot_air_fresh = m_dot_comp_in - m_dot_egr

h_air_amb = Enthalpy(Air,T=T_amb)
h_egr     = Enthalpy(Air,T=T_egr)

h_comp_in = (m_dot_air_fresh*h_air_amb + m_dot_egr*h_egr)/m_dot_comp_in
T_comp_in = Temperature(Air,h=h_comp_in)

"Compressor"
P_comp_in = P_amb
s_comp_in = Entropy(Air,T=T_comp_in,P=P_comp_in)
T_comp_out_is = Temperature(Air,s=s_comp_in,P=P_comb)

h_comp_in_air  = Enthalpy(Air,T=T_comp_in)
h_comp_out_is  = Enthalpy(Air,T=T_comp_out_is)
W_comp_spec    = (h_comp_out_is - h_comp_in_air)/eta_comp
h_comp_out     = h_comp_in_air + W_comp_spec
T_comp_out     = Temperature(Air,h=h_comp_out)
W_comp_total   = m_dot_comp_in*W_comp_spec

"Combustor (TIT clamp / design input)"
T_44 = 1404 [C]
h_44 = Enthalpy(Air,T=T_44)

Q_comb_in  = m_dot_comp_in*h_comp_out + m_dot_syngas*(h_syngas + eta_comb*LHV_syngas)
Q_comb_out = m_dot_comb_out*h_44

"Gas turbine"
s_turb_in    = Entropy(Air,T=T_44,P=P_comb)
T_45_is      = Temperature(Air,s=s_turb_in,P=P_amb)
h_45_is      = Enthalpy(Air,T=T_45_is)
W_turb_spec  = (h_44 - h_45_is)*eta_turb
h_45         = h_44 - W_turb_spec
T_45         = Temperature(Air,h=h_45)
W_turb_total = m_dot_comb_out*W_turb_spec

W_GT_Net = W_turb_total - W_comp_total

"--------------------------------------------------------------------------"
" 2) REAL GAS MIXTURE PROPERTIES FOR VALIDATION"
"--------------------------------------------------------------------------"

P_comp_in = P_amb

hN2_ref  = Enthalpy(Nitrogen,T=T_amb)
hO2_ref  = Enthalpy(Oxygen,T=T_amb)
hCO2_ref = Enthalpy(CarbonDioxide,T=T_amb)
hH2O_ref = Enthalpy(Steam,T=T_amb,P=P_amb)
hAr_ref  = Enthalpy(Argon,T=T_amb,P=P_amb)
hCO_ref  = Enthalpy(CO,T=T_amb)

h_comp_in_real = mf_N2_flue*Enthalpy(Nitrogen,T=T_comp_in) + mf_O2_flue*Enthalpy(Oxygen,T=T_comp_in) + mf_CO2_flue*Enthalpy(CarbonDioxide,T=T_comp_in) + mf_H2O_flue*Enthalpy(Steam,T=T_comp_in,P=P_amb) + mf_Ar_flue*Enthalpy(Argon,T=T_comp_in,P=P_amb)

h_45_real = mf_N2_flue*(Enthalpy(Nitrogen,T=T_45)-hN2_ref) + mf_O2_flue*(Enthalpy(Oxygen,T=T_45)-hO2_ref) + mf_CO2_flue*(Enthalpy(CarbonDioxide,T=T_45)-hCO2_ref) + mf_H2O_flue*(Enthalpy(Steam,T=T_45,P=P_amb)-hH2O_ref) + mf_Ar_flue*(Enthalpy(Argon,T=T_45,P=P_amb)-hAr_ref) + mf_H2_flue*Enthalpy(H2,T=T_45) + mf_CO_flue*(Enthalpy(CO,T=T_45)-hCO_ref)

Cp_gas_duct = mf_N2_flue*Cp(Nitrogen,T=500,P=P_amb) + mf_O2_flue*Cp(Oxygen,T=500,P=P_amb) + mf_CO2_flue*Cp(CarbonDioxide,T=500,P=P_amb) + mf_H2O_flue*Cp(Steam,T=500,P=P_amb) + mf_Ar_flue*Cp(Argon,T=500,P=P_amb)

"--------------------------------------------------------------------------"
" 3) SOLAR FIELD / MOLTEN SALT / SOLAR DUTIES"
"--------------------------------------------------------------------------"

m_dot_salt = Q_solar_available/(Cp_salt*(T_salt_hot - T_salt_cold))

h_cond_out = Enthalpy(Steam,P=P_condenser,x=0)

W_spec_pump1 = v_cond_out*(P_IP - P_condenser)/eta_pump
h_pump1_out = h_cond_out + W_spec_pump1

" 5) RANKINE CYCLE (MAIN POWER STREAM)"
